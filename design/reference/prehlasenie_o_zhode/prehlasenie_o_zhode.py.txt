PRAZDNE_RIADKY = 5  # počet prázdnych riadkov v šablóne bez produktov


def _e(t):
    return html.escape(str(t or ""))


def _data_uri(cesta):
    if not cesta:
        return ""
    p = Path(cesta)
    if not p.exists():
        raise FileNotFoundError(f"Súbor neexistuje: {p}")
    mime = mimetypes.guess_type(p.name)[0] or "image/png"
    return f"data:{mime};base64," + base64.b64encode(p.read_bytes()).decode()


def _logo(biele=False):
    return (ADRESAR / ("logo_white.svg" if biele else "logo.svg")).read_text(encoding="utf-8").split("?>", 1)[-1]


def _tabulka(produkty):
    stlpce = [("nazov", "Názov výrobku")]
    for k, nazov in (("kod", "Kód"), ("norma", "Norma"), ("poznamka", "Poznámka")):
        if not produkty or any(p.get(k) for p in produkty):
            stlpce.append((k, nazov))
    hlava = "".join(f'<th class="c-{k}">{n}</th>' for k, n in stlpce)
    riadky = []
    if produkty:
        for i, p in enumerate(produkty, 1):
            bunky = "".join(f'<td class="c-{k}">{_e(p.get(k))}</td>' for k, _ in stlpce)
            riadky.append(f'<tr><td class="c-i">{i}</td>{bunky}</tr>')
    else:
        for i in range(1, PRAZDNE_RIADKY + 1):
            bunky = "".join(f'<td class="c-{k}"></td>' for k, _ in stlpce)
            riadky.append(f'<tr class="blank"><td class="c-i">{i}</td>{bunky}</tr>')
    return f'<table class="prod"><thead><tr><th class="c-i">#</th>{hlava}</tr></thead><tbody>{"".join(riadky)}</tbody></table>'

    odb = ""

def _footer():
    ico = f'IČO {FIRMA["ico"]} · ' if FIRMA["ico"] else ""
    return f"""<div style="width:100%;font-family:Inter,sans-serif;font-size:7pt;color:#6B7891;padding:0 18mm;
      display:flex;justify-content:space-between;align-items:center;-webkit-print-color-adjust:exact;">
      <span><b style="color:#14264A">{_e(FIRMA["nazov"])}</b> · {_e(FIRMA["ulica"])}, {_e(FIRMA["mesto"])} ·
      {ico}IČ DPH {_e(FIRMA["ic_dph"])} · {_e(FIRMA["web"])}</span>
      <span>Strana <span class="pageNumber"></span> / <span class="totalPages"></span></span></div>"""


def vytvor_pdf(data, vystup):
    from playwright.sync_api import sync_playwright
    with sync_playwright() as p:
        prehliadac = p.chromium.launch()
        strana = prehliadac.new_page()
        strana.set_content(zostav_html(data), wait_until="load")
        strana.pdf(path=str(vystup), format="A4", print_background=True, prefer_css_page_size=True,
                   display_header_footer=True, header_template="<span></span>", footer_template=_footer())
        prehliadac.close()
    return Path(vystup)
